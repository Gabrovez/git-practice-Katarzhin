git version - Если Git установлен корректно, в терминале отобразится его версия
git config --global user.name "ваш_ник" - меняет глобального юзера 
git config --global user.email "ваша_почта@yandex.ru" - меняет глобальную почту
mkdir "название папки" - создать папку 
cd .. / "Название папки" - перейти в папку
git init -b main - создать локальный репозиторий и задать что бы он был main а не master
git remote add origin ссылка_на_репозиторий - привязять локальный репозиторий к удалённому
git remote -v - В результате будут отображены адреса удалённого репозитория для операций получения (fetch) и отправки (push) изменений
git status - проверить состояние репозитория
git add . - добавить все файлы (кроме игнорируемых) в индекс
git commit -m "сообщение" - создать коммит с сообщением
git log - команда для показа истории коммитов в текущей ветке
git push -u origin main - Отправить изменения в удалённый репозиторий
git log --oneline - компактеый git log
git show ваш_хеш - подробный разбор конкретного коммита
git checkout -b/ switch -c "Название новой ветки" - создать и сразу переключиться на новую ветку
git branch - просмотр всех веток
git switch " " - переключение на разные ветки
git pull - взять измененния с удалённого репозитория
